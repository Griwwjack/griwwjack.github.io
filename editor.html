// Lógica de Salvar CORRIGIDA document.getElementById('btnSave').onclick = async () => { const content = editor.value; const currentName = fileNameDisplay.textContent.replace(" (recuperado)", ""); try { // Tenta usar a API moderna (Chrome/Edge/Opera) if ('showSaveFilePicker' in window) { const handle = await window.showSaveFilePicker({ suggestedName: currentName, types: [{ description: 'Txt', accept: {'text/plain': ['.txt']} }] }); // Se chegou aqui, o usuário NÃO cancelou const writable = await handle.createWritable(); await writable.write(content); await writable.close(); updateEditor(content, handle.name); } else { // Se o navegador não suporta a API (Firefox/Safari/Mobile antigo) throw new Error('Modern API not supported'); } } catch (e) { // Se o erro for porque o usuário cancelou, não fazemos nada e saímos da função if (e.name === 'AbortError') { console.log('Salvamento cancelado pelo usuário.'); return; } // Se o erro for outro (como falta de suporte), aí sim rodamos o Fallback de download const blob = new Blob([content], { type: 'text/plain' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = currentName; a.click(); URL.revokeObjectURL(url); } };
